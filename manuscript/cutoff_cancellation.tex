\title{A Cutoff Criterion for Cancellation on the Plane}
\author{}
\date{}
\begin{document}
\maketitle
\begin{abstract}
We record a cancellation criterion for an integrable function on the plane. A uniform bound on the testing functions and an integrable flux-tail estimate imply that the source has zero mean. The criterion isolates a limiting argument relevant to weighted elliptic estimates; verifying its hypotheses for a compressible flow remains a separate problem.
\end{abstract}
\section{The cancellation criterion}
All integrals are taken with respect to Lebesgue measure on $\R^2$, and $|x|$ denotes the Euclidean norm. The cancellation argument in Fan--L\"u--Wang \cite[Lemma 3.3]{FLW} motivates the following elementary analytic criterion.
\begin{lemma}\label{lem:cutoff}
Let $f,h\in L^1(\R^2)$ be real-valued, with $h\ge0$ almost everywhere. Let $\chi_n:\R^2\to\R$, $n\in\mathbb N$, be measurable. Suppose that constants $B,C\ge0$ satisfy
\begin{equation}\label{eq:cutoffs}
 |\chi_n(x)|\le B\quad(n\in\mathbb N,\ x\in\R^2),
 \qquad \chi_n(x)\longrightarrow1\quad\text{for a.e. }x.
\end{equation}
If, for every $n\in\mathbb N$,
\begin{equation}\label{eq:flux}
 \left|\int_{\R^2} f\chi_n\dd\right|
 \le C\left(\int_{\{|x|\ge n\}}h\dd\right)^{1/2},
\end{equation}
then $\int_{\R^2}f\dd=0$.
\end{lemma}
\begin{proof}
Put $E_n=\{x\in\R^2:|x|\ge n\}$ and $H_n=\int_{E_n}h\dd$.
Each $E_n$ is measurable. Since $h\in L^1(\R^2)$, its restriction to $E_n$ is integrable, and $H_n\ge0$. For each fixed $x$, the indicator $\mathbf1_{E_n}(x)$ is eventually zero. Hence
\[
 h\mathbf1_{E_n}\longrightarrow0,
 \qquad |h\mathbf1_{E_n}|\le|h|\in L^1(\R^2).
\]
The dominated convergence theorem gives
\begin{equation}\label{eq:tail}
 H_n=\int_{\R^2}h\mathbf1_{E_n}\dd\longrightarrow0.
\end{equation}
On the other hand, each $f\chi_n$ is measurable up to a null set and
\[
 |f\chi_n|\le B|f|\in L^1(\R^2).
\]
Thus all testing integrals in \eqref{eq:flux} are integrals of integrable functions. By \eqref{eq:cutoffs}, $f\chi_n\to f$ almost everywhere. A second application of dominated convergence yields
\begin{equation}\label{eq:testlimit}
 I_n:=\int_{\R^2}f\chi_n\dd\longrightarrow
 I:=\int_{\R^2}f\dd.
\end{equation}
Combining \eqref{eq:flux} and \eqref{eq:tail}, we obtain
\[
 0\le |I_n|\le C\sqrt{H_n}\longrightarrow0.
\]
Equation \eqref{eq:testlimit} also implies $|I_n|\to|I|$. Uniqueness of the limit gives $|I|=0$, as asserted.
\end{proof}
\begin{remark}
The estimate \eqref{eq:flux} is a substantive hypothesis. An application to a momentum equation must establish it from admissible testing, integrability, and control of the flux at infinity. Lemma~\ref{lem:cutoff} does not assert that these requirements hold for a particular compressible solution, nor does it establish a global existence theorem.
\end{remark}
